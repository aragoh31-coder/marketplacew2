#include "core/or/or.h"
#include "lib/crypt_ops/crypto_rand.h"
#include "lib/crypt_ops/crypto_digest.h"
#include "lib/crypt_ops/crypto_argon2.h"
#include "lib/encoding/binascii.h"
#include <time.h>
#include <string.h>
#include <stdlib.h>

#define POW_EQUIX   0
#define POW_ARGON2  1

typedef struct {
    uint8_t algorithm;      /* 0=Equi-X, 1=Argon2 */
    uint8_t difficulty;     /* leading-zero bits required */
    uint8_t nonce[16];      /* random */
    char    solution[256];  /* client-computed */
} pow_challenge_t;

/* Generate a new challenge */
void tor_pow_generate(pow_challenge_t *c) {
    c->algorithm = (rand() & 1) ? POW_ARGON2 : POW_EQUIX;
    c->difficulty = 20 + (rand() % 12); /* 20–31 bits */
    crypto_rand(c->nonce, sizeof(c->nonce));
    memset(c->solution, 0, sizeof(c->solution));
}

/* Verify Equi-X: count leading-zero bits in SHA256(nonce||solution) */
static int verify_equix(const pow_challenge_t *c) {
    uint8_t hash[32];
    crypto_digest256(hash,
                     c->nonce, sizeof(c->nonce),
                     DIGEST_SHA256_CTX_SHA256);
    crypto_digest256(hash,
                     (uint8_t*)c->solution, strlen(c->solution),
                     DIGEST_SHA256_FINAL_SHA256);
    int zeros = 0;
    for (int i = 0; i < 32; i++) {
        if (hash[i] == 0) { zeros += 8; continue; }
        for (int b = 7; b >= 0; b--) {
            if (!(hash[i] & (1 << b))) zeros++;
            else break;
        }
        break;
    }
    return zeros >= c->difficulty;
}

/* Verify Argon2id: memory-hard puzzle */
static int verify_argon2(const pow_challenge_t *c) {
    uint8_t out[32];
    struct Argon2_Context ctx = {
        .out        = out,
        .outlen     = 32,
        .pwd        = (uint8_t*)c->solution,
        .pwdlen     = strlen(c->solution),
        .salt       = c->nonce,
        .saltlen    = sizeof(c->nonce),
        .t_cost     = (c->difficulty/8)+2,
        .m_cost     = 1<<14,
        .lanes      = 1,
        .threads    = 1,
        .version    = ARGON2_VERSION_13,
        .allocate_cbk = NULL,
        .free_cbk     = NULL,
        .flags        = ARGON2_DEFAULT_FLAGS
    };
    return argon2_ctx(&ctx, Argon2_i) == ARGON2_OK;
}

/* Public API: verify challenge */
int tor_pow_verify(const pow_challenge_t *c) {
    return c->algorithm == POW_EQUIX
         ? verify_equix(c)
         : verify_argon2(c);
}

#include "core/or/or.h"
#include "lib/crypt_ops/crypto_hmac.h"
#include "lib/encoding/base64.h"
#include "lib/crypt_ops/crypto_rand.h"
#include <time.h>
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

#define HMAC_KEY_LEN   32
#define HMAC_SIG_LEN   32
#define TOKEN_LIFETIME 7200   /* 2 hours */
#define MIN_SOLVE_TIME   3   /* seconds */
#define MAX_SOLVE_TIME 300   /* seconds */
#define HONEYPOT_COUNT   5

static uint8_t hmac_key[HMAC_KEY_LEN];
static time_t last_key_rotation = 0;

typedef struct {
    uint32_t magic;         /* 0x544F5248 "TORH" */
    time_t   issued;        /* issuance time */
    time_t   expires;       /* expiry time */
    uint32_t remaining;     /* requests remaining */
    uint8_t  sig[HMAC_SIG_LEN];
} hmac_token_t;

typedef struct {
    uint8_t  id[16];
    time_t   created;
    time_t   mint;          /* min solve time */
    time_t   maxt;          /* max solve time */
    uint32_t a, b, expected;
    int      difficulty;
    char     honeypots[HONEYPOT_COUNT][32];
    uint8_t  hmac[HMAC_SIG_LEN];
} hmac_challenge_t;

/* Initialize/rotate HMAC key daily */
static void hmac_init(void) {
    time_t now = time(NULL);
    if (!last_key_rotation || now - last_key_rotation > 86400) {
        crypto_rand(hmac_key, HMAC_KEY_LEN);
        last_key_rotation = now;
    }
}

/* Sign data */
static void hmac_sign(const void *data, size_t len, uint8_t *out) {
    crypto_hmac_sha256((char*)out, (const char*)hmac_key, HMAC_KEY_LEN,
                       data, len);
}

/* Create new challenge */
hmac_challenge_t* create_challenge(int diff) {
    hmac_init();
    hmac_challenge_t *c = malloc(sizeof(*c));
    memset(c,0,sizeof(*c));
    crypto_rand(c->id, 16);
    c->created = time(NULL);
    c->mint    = c->created + MIN_SOLVE_TIME;
    c->maxt    = c->created + MAX_SOLVE_TIME;
    c->difficulty = diff;
    if (diff < 2) {
        c->a = rand()%50 + 1; c->b = rand()%50 + 1;
        c->expected = c->a + c->b;
    } else {
        c->a = rand()%10 + 1; c->b = rand()%10 + 1;
        c->expected = c->a * c->b;
    }
    for(int i=0;i<HONEYPOT_COUNT;i++){
        snprintf(c->honeypots[i],32,"hp%02d_%02x",i,hmac_key[i]);
    }
    hmac_sign(c, sizeof(*c)-HMAC_SIG_LEN, c->hmac);
    return c;
}

/* Verify challenge */
int verify_challenge(const hmac_challenge_t *c, const char *ans,
                     time_t start, const char *hp_vals[]) {
    time_t now = time(NULL);
    if (start < c->mint || now > c->maxt) return 0;
    for(int i=0;i<HONEYPOT_COUNT;i++){
        if (hp_vals[i] && hp_vals[i][0]) return 0;
    }
    uint8_t sig[HMAC_SIG_LEN];
    hmac_sign(c, sizeof(*c)-HMAC_SIG_LEN, sig);
    if (memcmp(sig, c->hmac, HMAC_SIG_LEN)) return 0;
    if ((uint32_t)atoi(ans) != c->expected) return 0;
    return 1;
}

/* Issue token on success */
hmac_token_t* issue_token(void) {
    hmac_init();
    hmac_token_t *t = malloc(sizeof(*t));
    t->magic     = 0x544F5248;
    t->issued    = time(NULL);
    t->expires   = t->issued + TOKEN_LIFETIME;
    t->remaining = 500;
    hmac_sign(t, sizeof(*t)-HMAC_SIG_LEN, t->sig);
    return t;
}

/* Verify and consume token */
int verify_token(hmac_token_t *t) {
    time_t now = time(NULL);
    if (t->magic!=0x544F5248 || now>t->expires || t->remaining==0) return 0;
    uint8_t sig[HMAC_SIG_LEN];
    hmac_sign(t, sizeof(*t)-HMAC_SIG_LEN, sig);
    if (memcmp(sig, t->sig, HMAC_SIG_LEN)) return 0;
    t->remaining--;
    hmac_sign(t, sizeof(*t)-HMAC_SIG_LEN, t->sig);
    return 1;
}

from django.utils.deprecation import MiddlewareMixin
from django.shortcuts import redirect
from .utils import decode_token, verify_token

class AntiDDoSMiddleware(MiddlewareMixin):
    def process_request(self, request):
        if request.path.startswith('/anti_ddos/'): return None
        token = request.COOKIES.get('hmac_token')
        if token and verify_token(decode_token(token)):
            return None
        return redirect('anti_ddos:challenge')

def verify_view(request):
    data = request.POST.get('challenge_data')
    ans = request.POST.get('answer')
    start = int(request.POST.get('start_time'))
    hps = [request.POST.get(f'hp{i}') for i in range(5)]
    c = decode_challenge(data)
    if verify_challenge(c, ans, start, hps):
        t = issue_token()
        resp = redirect('/')
        resp.set_cookie('hmac_token', serialize_token(t),
                        max_age=7200, httponly=True, secure=True)
        return resp
    return redirect('anti_ddos:challenge')

    obj = json.loads(base64.b64decode(s).decode())

    t.expires = obj['expires']
    t.remaining = obj['remaining']

#!/usr/bin/env python3
import hashlib, time, random, re, base64
import requests
from stem import Signal
from stem.control import Controller

# Tor controller
with Controller.from_port(port=9051) as ctrl:
    ctrl.authenticate()
    ctrl.signal(Signal.NEWNYM)

session = requests.Session()
session.proxies.update({'http':'socks5h://localhost:9050','https':'socks5h://localhost:9050'})

def fetch_challenge():
    r = session.get('http://your-onion.onion/anti_ddos/challenge/')
    html = r.text
    data = re.search(r'name="challenge_data" value="([^"]+)"', html).group(1)
    start = re.search(r'name="start_time" value="([^"]+)"', html).group(1)
    return data, int(start)

def solve_math(html):
    match = re.search(r'>(\\d+) [×+] (\\d+) =', html)
    if match:
        a, b = map(int, match.groups())
        return str(a*b) if '×' in html else str(a+b)
    return '0'

def main():
    data, start = fetch_challenge()
    r = session.get('http://your-onion.onion/anti_ddos/challenge/')
    sol = solve_math(r.text)
    session.post('http://your-onion.onion/anti_ddos/verify/', data={
        'challenge_data': data,
        'start_time': str(start),
        'answer': sol,
        **{'hp'+str(i):'' for i in range(5)}
    })
    print("Submitted solution.")

if __name__=="__main__":
    main()

local cjson = require "cjson"
local hmac  = require "resty.hmac"
local str   = require "resty.string"

local SECRET = "super_secret_key_32bytes"

local function verify_token(tok)
  if not tok then return false end
  local raw = ngx.decode_base64(tok)
  local ok, obj = pcall(cjson.decode, raw)
  if not ok then return false end
  local sig = obj.sig; obj.sig = nil
  local payload = cjson.encode(obj)
  local hm, err = hmac:new(SECRET, hmac.ALGOS.SHA256)
  if not hm then return false end
  local expected = str.to_hex(hm:final(payload))
  return expected == sig
end

if not verify_token(ngx.var.cookie_hmac_token) then
  return ngx.redirect("/anti_ddos/challenge/")
end

        path('anti_ddos/', include('apps.anti_ddos.urls'))
   - Ensure templates in `templates/anti_ddos/`

   - No JavaScript is used anywhere

   - Install: pip install stem requests
   - Configure Tor control port in torrc (ControlPort 9051, HashedControlPassword)
   - Run `tor_pow_launcher.py` as a service
